## 一、核心概念回顾

### 1.1 什么是扫码登录？

**扫码登录的工作流程**:
```
1. 客户端请求登录页面
   └─> 服务器生成QR码 (包含随机token/sessionId)

2. QR码显示在浏览器上
   └─> 用户用手机扫描QR码

3. 手机端验证身份并确认登录
   └─> 向服务器发送确认请求

4. 服务器验证确认信息
   └─> 更新session状态为已认证

5. 浏览器端检测到登录完成
   └─> 浏览器自动跳转到首页或目标页面
```

### 1.2 无头浏览器在扫码中的作用

| 组件 | 职责 |
|------|------|
| **QR码生成** | 服务器端负责，浏览器显示 |
| **QR码检测** | 无头浏览器通过CSS选择器检测QR码出现 |
| **状态监听** | 监听URL变化或DOM变化来检测登录完成 |
| **Cookie收集** | 登录完成后自动收集Session Cookie |
| **会话复用** | 持久化Cookie，后续请求直接使用 |

## 二、Playwright实现 - 扫码登录

### 2.1 基础实现：等待QR码和登录完成

```python
import asyncio
from playwright.async_api import async_playwright, Page, Browser
from typing import Optional, Dict, Any
import json
from datetime import datetime
import logging

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class QRCodeLoginHandler:
    """处理扫码登录流程"""

    def __init__(self, headless: bool = False):
        """
        初始化登录处理器

        Args:
            headless: 是否以无头模式运行（生产环境True，调试False）
        """
        self.browser: Optional[Browser] = None
        self.page: Optional[Page] = None
        self.headless = headless
        self.cookies = {}

    async def init(self):
        """初始化浏览器"""
        self.playwright = await async_playwright().start()
        self.browser = await self.playwright.chromium.launch(
            headless=self.headless,
            args=[
                '--disable-blink-features=AutomationControlled',  # 隐藏自动化特征
                '--no-sandbox',
                '--disable-setuid-sandbox'
            ]
        )
        logger.info("浏览器已启动")

    async def cleanup(self):
        """清理资源"""
        if self.page:
            await self.page.close()
        if self.browser:
            await self.browser.close()
        if self.playwright:
            await self.playwright.stop()
        logger.info("浏览器已关闭")

    async def navigate_to_login(self, login_url: str) -> bool:
        """
        导航到登录页面

        Args:
            login_url: 登录页面URL

        Returns:
            是否成功导航
        """
        try:
            if not self.page:
                self.page = await self.browser.new_page()

            response = await self.page.goto(login_url, wait_until='networkidle')
            logger.info(f"已导航到登录页: {login_url}")
            return response.status == 200

        except Exception as e:
            logger.error(f"导航失败: {str(e)}")
            return False

    async def wait_for_qrcode(self, qr_selector: str, timeout_ms: int = 10000) -> bool:
        """
        等待QR码出现

        Args:
            qr_selector: QR码容器的CSS选择器
            timeout_ms: 等待超时时间（毫秒）

        Returns:
            QR码是否出现
        """
        try:
            await self.page.wait_for_selector(qr_selector, timeout=timeout_ms)
            logger.info(f"✓ QR码已出现 (选择器: {qr_selector})")
            return True

        except Exception as e:
            logger.error(f"✗ QR码等待超时: {str(e)}")
            return False

    async def wait_for_login_completion(
        self,
        check_method: str = "url_change",
        timeout_ms: int = 60000
    ) -> bool:
        """
        等待登录完成

        Three methods to detect login completion:
        1. URL change - 页面URL变化表示登录成功
        2. DOM change - 特定DOM元素出现表示登录成功
        3. Cookie check - 检测特定Cookie出现

        Args:
            check_method: 检测方法 ("url_change", "dom_change", "cookie_check")
            timeout_ms: 超时时间

        Returns:
            是否成功登录
        """
        try:
            initial_url = self.page.url
            start_time = datetime.now()

            if check_method == "url_change":
                return await self._wait_url_change(initial_url, timeout_ms)

            elif check_method == "dom_change":
                # 需要指定登录成功后出现的DOM选择器
                success_selector = ".home-container, .dashboard, [role='main']"
                return await self._wait_dom_change(success_selector, timeout_ms)

            elif check_method == "cookie_check":
                return await self._wait_cookie_change(timeout_ms)

        except Exception as e:
            logger.error(f"登录完成检测失败: {str(e)}")
            return False

    async def _wait_url_change(self, initial_url: str, timeout_ms: int) -> bool:
        """等待URL变化（通常表示重定向到首页）"""
        start_time = datetime.now()

        while True:
            current_url = self.page.url
            if current_url != initial_url:
                logger.info(f"✓ URL已变化: {initial_url} -> {current_url}")
                return True

            elapsed = (datetime.now() - start_time).total_seconds() * 1000
            if elapsed > timeout_ms:
                logger.error("✗ URL检测超时")
                return False

            await asyncio.sleep(0.5)  # 每500ms检查一次

    async def _wait_dom_change(self, success_selector: str, timeout_ms: int) -> bool:
        """等待特定DOM元素出现"""
        try:
            await self.page.wait_for_selector(success_selector, timeout=timeout_ms)
            logger.info(f"✓ 登录成功标记已出现: {success_selector}")
            return True
        except Exception as e:
            logger.error(f"✗ DOM检测失败: {str(e)}")
            return False

    async def _wait_cookie_change(self, timeout_ms: int) -> bool:
        """等待特定Cookie出现"""
        start_time = datetime.now()
        target_cookie_names = ["session_id", "auth_token", "sid"]

        while True:
            try:
                cookies = await self.page.context.cookies()
                cookie_names = [c["name"] for c in cookies]

                # 检查是否有目标Cookie
                for target in target_cookie_names:
                    if target in cookie_names:
                        logger.info(f"✓ 认证Cookie已出现: {target}")
                        return True

                elapsed = (datetime.now() - start_time).total_seconds() * 1000
                if elapsed > timeout_ms:
                    logger.error("✗ Cookie检测超时")
                    return False

                await asyncio.sleep(1)

            except Exception as e:
                logger.error(f"Cookie检测异常: {str(e)}")
                return False

    async def extract_cookies(self) -> Dict[str, str]:
        """
        提取登录后的Cookie

        Returns:
            Cookie字典 {name: value}
        """
        try:
            cookies = await self.page.context.cookies()
            self.cookies = {c["name"]: c["value"] for c in cookies}
            logger.info(f"✓ 已提取 {len(self.cookies)} 个Cookie")
            return self.cookies

        except Exception as e:
            logger.error(f"Cookie提取失败: {str(e)}")
            return {}

    async def save_cookies(self, filepath: str):
        """保存Cookie到文件"""
        try:
            with open(filepath, 'w') as f:
                json.dump(self.cookies, f, indent=2)
            logger.info(f"✓ Cookie已保存到: {filepath}")

        except Exception as e:
            logger.error(f"Cookie保存失败: {str(e)}")

    async def login_with_qrcode(
        self,
        login_url: str,
        qr_selector: str,
        success_indicator: str = "url_change"
    ) -> bool:
        """
        完整的扫码登录流程

        Args:
            login_url: 登录页面URL
            qr_selector: QR码CSS选择器
            success_indicator: 登录成功指标 ("url_change", "dom_change", "cookie_check")

        Returns:
            登录是否成功
        """
        try:
            await self.init()

            # Step 1: 导航到登录页
            if not await self.navigate_to_login(login_url):
                return False

            # Step 2: 等待QR码出现
            if not await self.wait_for_qrcode(qr_selector):
                return False

            logger.info("📱 请用手机扫描QR码进行登录...")

            # Step 3: 等待登录完成
            if not await self.wait_for_login_completion(success_indicator):
                return False

            # Step 4: 提取Cookie
            await self.extract_cookies()

            logger.info("✓ 登录流程完成")
            return True

        except Exception as e:
            logger.error(f"登录流程异常: {str(e)}")
            return False

        finally:
            await self.cleanup()


# 使用示例
async def main():
    """测试扫码登录"""
    handler = QRCodeLoginHandler(headless=False)  # 调试时设为False看浏览器

    success = await handler.login_with_qrcode(
        login_url="https://fsdocs.sf-express.com/login",
        qr_selector="#qr-code, .qr-code-container, [data-testid='qr-code']",
        success_indicator="url_change"
    )

    if success:
        # 保存Cookie供后续使用
        await handler.save_cookies("cookies.json")
        print("✓ 登录成功，Cookie已保存")
    else:
        print("✗ 登录失败")


if __name__ == "__main__":
    asyncio.run(main())
```

### 2.2 高级实现：监听WebSocket事件

有些应用通过WebSocket实时推送登录状态，这样比轮询更高效：

```python
import asyncio
from playwright.async_api import Page, WebSocketFrame
import json
from typing import Optional, Callable


class WebSocketLoginListener:
    """通过WebSocket监听登录完成事件"""

    def __init__(self, page: Page):
        self.page = page
        self.login_complete = False
        self.websocket = None

    async def listen_for_login(self, timeout_ms: int = 60000) -> bool:
        """
        监听WebSocket消息来检测登录完成

        Args:
            timeout_ms: 超时时间

        Returns:
            是否检测到登录完成
        """

        def on_websocket_frame(frame: WebSocketFrame):
            """WebSocket消息处理"""
            try:
                # 尝试解析JSON消息
                data = json.loads(frame.payload)

                # 检测登录成功消息
                if data.get("type") == "login_complete":
                    self.login_complete = True
                    logger.info("✓ 通过WebSocket检测到登录完成")

                elif data.get("status") == "authenticated":
                    self.login_complete = True
                    logger.info("✓ 通过WebSocket检测到认证成功")

                else:
                    logger.debug(f"WebSocket消息: {data}")

            except json.JSONDecodeError:
                # 不是JSON格式的消息，跳过
                pass
            except Exception as e:
                logger.error(f"WebSocket处理异常: {str(e)}")

        # 注册WebSocket事件监听器
        self.page.on("websocket", lambda ws: ws.on("framereceived", on_websocket_frame))

        # 等待登录完成或超时
        start_time = asyncio.get_event_loop().time()
        while not self.login_complete:
            elapsed = (asyncio.get_event_loop().time() - start_time) * 1000

            if elapsed > timeout_ms:
                logger.error("✗ WebSocket监听超时")
                return False

            await asyncio.sleep(0.1)

        return True


# 使用示例
async def login_with_websocket_listener():
    """使用WebSocket监听器的登录"""
    handler = QRCodeLoginHandler(headless=False)

    try:
        await handler.init()
        await handler.navigate_to_login("https://fsdocs.sf-express.com/login")
        await handler.wait_for_qrcode("#qr-code")

        # 使用WebSocket监听器
        ws_listener = WebSocketLoginListener(handler.page)
        success = await ws_listener.listen_for_login()

        if success:
            await handler.extract_cookies()
            await handler.save_cookies("cookies.json")
            return True
        else:
            return False

    finally:
        await handler.cleanup()
```

## 三、会话管理和持久化

### 3.1 Cookie持久化和复用

```python
import json
from pathlib import Path
from typing import Dict, List


class SessionManager:
    """会话和Cookie管理"""

    def __init__(self, cookie_file: str = "cookies.json"):
        self.cookie_file = Path(cookie_file)

    def save_session(self, cookies: Dict[str, str], metadata: Dict = None):
        """
        保存会话Cookie到文件

        Args:
            cookies: Cookie字典
            metadata: 额外的元数据（如过期时间、获取时间等）
        """
        session_data = {
            "cookies": cookies,
            "timestamp": datetime.now().isoformat(),
            "metadata": metadata or {}
        }

        with open(self.cookie_file, 'w') as f:
            json.dump(session_data, f, indent=2)

        logger.info(f"✓ 会话已保存: {self.cookie_file}")

    def load_session(self) -> Optional[Dict[str, str]]:
        """
        从文件加载Cookie

        Returns:
            Cookie字典，如果文件不存在返回None
        """
        if not self.cookie_file.exists():
            logger.warning(f"会话文件不存在: {self.cookie_file}")
            return None

        try:
            with open(self.cookie_file, 'r') as f:
                session_data = json.load(f)

            cookies = session_data.get("cookies", {})
            timestamp = session_data.get("timestamp")
            logger.info(f"✓ 会话已加载 (保存于: {timestamp})")
            return cookies

        except Exception as e:
            logger.error(f"会话加载失败: {str(e)}")
            return None

    async def apply_cookies_to_page(self, page: Page, cookies: Dict[str, str]):
        """
        将Cookie应用到页面上下文

        Args:
            page: Playwright Page对象
            cookies: Cookie字典
        """
        try:
            # 转换为Playwright格式
            cookie_list = [
                {
                    "name": name,
                    "value": value,
                    "domain": ".sf-express.com",
                    "path": "/",
                    "expires": -1,  # Session cookie
                    "httpOnly": False,
                    "secure": True,
                    "sameSite": "Lax"
                }
                for name, value in cookies.items()
            ]

            await page.context.add_cookies(cookie_list)
            logger.info(f"✓ 已应用 {len(cookie_list)} 个Cookie到页面")

        except Exception as e:
            logger.error(f"Cookie应用失败: {str(e)}")

    async def verify_session(self, test_url: str, cookies: Dict[str, str]) -> bool:
        """
        验证Cookie会话是否仍然有效

        Args:
            test_url: 用于验证的测试URL（需要认证才能访问）
            cookies: 要验证的Cookie

        Returns:
            会话是否有效
        """
        try:
            playwright = await async_playwright().start()
            browser = await playwright.chromium.launch(headless=True)
            page = await browser.new_page()

            # 应用Cookie
            await self.apply_cookies_to_page(page, cookies)

            # 尝试访问受保护的页面
            response = await page.goto(test_url, wait_until='networkidle')

            # 检查是否成功（404或403表示无效）
            is_valid = response.status not in [401, 403, 404]

            await page.close()
            await browser.close()
            await playwright.stop()

            logger.info(f"{'✓' if is_valid else '✗'} 会话有效性: {is_valid} (HTTP {response.status})")
            return is_valid

        except Exception as e:
            logger.error(f"会话验证异常: {str(e)}")
            return False


# 使用示例
async def login_flow_with_session_management():
    """完整的登录和会话管理流程"""
    session_mgr = SessionManager("sf_session.json")

    # 尝试加载已保存的会话
    cookies = session_mgr.load_session()

    if cookies:
        # 验证会话是否有效
        is_valid = await session_mgr.verify_session(
            "https://fsdocs.sf-express.com/desktop/mydoc/folder/test",
            cookies
        )

        if is_valid:
            logger.info("✓ 已有有效的会话，无需重新登录")
            return cookies

    # 需要重新登录
    logger.info("无有效会话，需要重新登录")
    handler = QRCodeLoginHandler()

    if await handler.login_with_qrcode(
        login_url="https://fsdocs.sf-express.com/login",
        qr_selector="#qr-code"
    ):
        # 保存新的会话
        session_mgr.save_session(handler.cookies)
        return handler.cookies
    else:
        logger.error("登录失败")
        return None
```

## 四、错误处理和重试机制

### 4.1 完整的错误处理策略

```python
from enum import Enum
import time


class LoginError(Exception):
    """登录相关的基础异常"""
    pass


class QRCodeNotFound(LoginError):
    """QR码未找到"""
    pass


class LoginTimeout(LoginError):
    """登录超时"""
    pass


class BrowserCrashError(LoginError):
    """浏览器崩溃"""
    pass


class RobustQRCodeLoginHandler(QRCodeLoginHandler):
    """带有错误恢复的鲁棒登录处理器"""

    def __init__(self, max_retries: int = 3, headless: bool = False):
        super().__init__(headless)
        self.max_retries = max_retries
        self.retry_count = 0

    async def login_with_retry(
        self,
        login_url: str,
        qr_selector: str,
        success_indicator: str = "url_change"
    ) -> bool:
        """
        带重试机制的登录

        Args:
            login_url: 登录页面URL
            qr_selector: QR码选择器
            success_indicator: 成功指标

        Returns:
            登录是否成功
        """

        for attempt in range(1, self.max_retries + 1):
            try:
                logger.info(f"--- 第 {attempt}/{self.max_retries} 次登录尝试 ---")

                success = await self.login_with_qrcode(
                    login_url, qr_selector, success_indicator
                )

                if success:
                    return True
                else:
                    logger.warning(f"第 {attempt} 次尝试失败，将重试")

            except BrowserCrashError as e:
                logger.error(f"浏览器崩溃: {str(e)}")
                # 清理并重新初始化
                try:
                    await self.cleanup()
                except:
                    pass

            except QRCodeNotFound as e:
                logger.error(f"QR码未找到: {str(e)}")

            except LoginTimeout as e:
                logger.error(f"登录超时: {str(e)}")

            except Exception as e:
                logger.error(f"未预期的错误: {str(e)}")

            # 重试前等待
            if attempt < self.max_retries:
                wait_time = min(2 ** attempt, 10)  # 指数退避，最多10秒
                logger.info(f"等待 {wait_time} 秒后重试...")
                await asyncio.sleep(wait_time)

        logger.error(f"所有 {self.max_retries} 次尝试都失败了")
        return False

    async def handle_qrcode_expiration(
        self,
        qr_selector: str,
        refresh_selector: str = ".refresh-qr, [data-action='refresh']"
    ) -> bool:
        """
        处理QR码过期的情况（通常需要刷新获取新的QR码）

        Args:
            qr_selector: 原始QR码选择器
            refresh_selector: 刷新按钮的选择器

        Returns:
            是否成功刷新
        """
        try:
            # 检查是否有"QR码已过期"的提示
            is_expired = await self.page.query_selector(".qr-expired, .expired-notice")

            if is_expired:
                logger.warning("⚠ QR码已过期，尝试刷新")

                # 点击刷新按钮
                refresh_btn = await self.page.query_selector(refresh_selector)
                if refresh_btn:
                    await refresh_btn.click()
                    await asyncio.sleep(1)

                    # 等待新的QR码出现
                    await self.wait_for_qrcode(qr_selector, timeout_ms=10000)
                    logger.info("✓ QR码已刷新")
                    return True
                else:
                    logger.error("未找到刷新按钮")
                    return False

            return True

        except Exception as e:
            logger.error(f"QR码刷新异常: {str(e)}")
            return False
```

## 五、性能优化

### 5.1 浏览器连接池

```python
from asyncio import Semaphore
from typing import List


class BrowserPool:
    """浏览器连接池，支持并发登录"""

    def __init__(self, max_size: int = 3):
        self.max_size = max_size
        self.semaphore = Semaphore(max_size)
        self.browsers: List[Browser] = []
        self.playwright = None

    async def init(self):
        """初始化浏览器池"""
        self.playwright = await async_playwright().start()
        logger.info(f"✓ 浏览器池已初始化 (大小: {self.max_size})")

    async def acquire(self) -> Browser:
        """获取一个浏览器实例"""
        await self.semaphore.acquire()

        try:
            browser = await self.playwright.chromium.launch(headless=True)
            self.browsers.append(browser)
            return browser
        except Exception as e:
            logger.error(f"浏览器启动失败: {str(e)}")
            self.semaphore.release()
            raise

    async def release(self, browser: Browser):
        """释放一个浏览器实例"""
        try:
            await browser.close()
            if browser in self.browsers:
                self.browsers.remove(browser)
        finally:
            self.semaphore.release()

    async def cleanup(self):
        """清理所有浏览器"""
        for browser in self.browsers:
            try:
                await browser.close()
            except:
                pass

        if self.playwright:
            await self.playwright.stop()

        logger.info("✓ 浏览器池已清理")


# 使用示例：并发登录多个账户
async def concurrent_logins(login_urls: List[str]):
    """并发执行多个登录"""
    pool = BrowserPool(max_size=3)
    await pool.init()

    async def login_task(login_url: str) -> bool:
        browser = await pool.acquire()
        try:
            handler = QRCodeLoginHandler()
            handler.browser = browser
            return await handler.login_with_qrcode(login_url, "#qr-code")
        finally:
            await pool.release(browser)

    tasks = [login_task(url) for url in login_urls]
    results = await asyncio.gather(*tasks)

    await pool.cleanup()
    return results
```

## 六、最佳实践总结

### 6.1 检查清单

- [ ] **使用隐身模式**：`--disable-blink-features=AutomationControlled` 隐藏自动化迹象
- [ ] **设置合理超时**：QR码等待10-15秒，登录完成等待60秒
- [ ] **持久化会话**：将Cookie保存到文件，下次复用
- [ ] **验证会话**：定期验证已保存的会话是否仍然有效
- [ ] **错误恢复**：实现重试机制和优雅降级
- [ ] **资源清理**：确保在异常情况下也能正确清理浏览器资源
- [ ] **日志记录**：详细记录每个步骤便于调试
- [ ] **性能优化**：对于批量操作使用浏览器池

### 6.2 常见问题排查

| 问题 | 原因 | 解决方案 |
|------|------|--------|
| QR码不显示 | 页面加载不完整 | 增加等待时间，检查CSS选择器 |
| 登录超时 | 二维码已过期 | 实现自动刷新机制 |
| Cookie无效 | 会话已过期 | 实现会话验证和重新登录 |
| 浏览器崩溃 | 内存泄漏 | 定期清理资源，使用浏览器池 |
| 反爬虫检测 | 自动化特征明显 | 使用隐身模式，添加随机延迟 |
