# 📁 第2-2层：标准项目结构模板

> **使命：** 设计3份可直接克隆的项目模板（小/中/大），集成S1-S6的Skill自动化
>
> **核心理念：** 新项目启动时，`git clone [template]` 就能得到一个标准化的、具有完整工具链的项目结构。所有Skill的运行脚本都已配置好，开发者只需专注于业务代码。

---

## 📊 项目模板总览

| 模板 | 适用场景 | 工期 | 项目复杂度 | Skill使用 | 目录深度 |
|------|--------|------|---------|---------|---------|
| **模板A：小项目** | 1-2人日的bug修复、小优化 | 5-7天 | 简单 | S1, S2, S3, S4 | 浅 |
| **模板B：中项目** | 5-15人日的常规功能 | 25天 | 中等 | S1-S6全部 | 中等 |
| **模板C：大项目** | 15+人日的核心功能 | 30-40天 | 复杂 | S1-S6全部 + 分模块 | 深 |

#### 3️⃣ `src/config/index.ts` - 配置加载（S4检查点）

```typescript
import dotenv from 'dotenv';
import { configSchema } from './env.schema';

// 加载环境变量
dotenv.config();

/**
 * 配置对象
 * 【重要】这个文件会被S4（配置检查Skill）自动扫描
 * 确保：
 * 1. 没有硬编码敏感信息（密码、API Key）
 * 2. 所有环境变量引用都正确
 * 3. 配置值都在预期范围内
 */

export const config = {
  server: {
    port: parseInt(process.env.PORT || '3000'),
    env: process.env.NODE_ENV || 'development'
  },

  database: {
    host: process.env.DB_HOST || 'localhost',
    port: parseInt(process.env.DB_PORT || '5432'),
    username: process.env.DB_USER || 'admin',
    password: process.env.DB_PASSWORD,  // 【必须】从环境变量读取，不能硬编码
    database: process.env.DB_NAME || 'myapp'
  },

  redis: {
    host: process.env.REDIS_HOST || 'localhost',
    port: parseInt(process.env.REDIS_PORT || '6379'),  // 【注意】REDIS_PORT拼写必须正确
    password: process.env.REDIS_PASSWORD  // 【必须】从环境变量读取
  },

  cache: {
    ttl: parseInt(process.env.CACHE_TTL || '3600')
  },

  logger: {
    level: process.env.LOG_LEVEL || 'info'
  }
};

/**
 * 【S4会检查这一段】
 * 验证关键配置是否合理
 */
export function validateConfig() {
  const errors: string[] = [];

  // 检查敏感信息是否硬编码
  if (!config.database.password) {
    errors.push('❌ 数据库密码未设置（DB_PASSWORD环境变量）');
  }

  // 检查端口范围
  if (config.server.port < 1 || config.server.port > 65535) {
    errors.push('❌ 服务端口不在有效范围（1-65535）');
  }

  // 检查Redis连接
  if (config.redis.port < 1 || config.redis.port > 65535) {
    errors.push('❌ Redis端口不在有效范围（1-65535）');
  }

  // 检查缓存TTL
  if (config.cache.ttl < 60 || config.cache.ttl > 86400) {
    errors.push('❌ 缓存TTL不在有效范围（60-86400秒）');
  }

  if (errors.length > 0) {
    console.error('❌ 配置检查失败：');
    errors.forEach(err => console.error(`  ${err}`));
    process.exit(1);
  }

  console.log('✅ 配置检查通过');
}

// 启动时验证配置
validateConfig();
```

#### 4️⃣ `scripts/skill-s4-check-config.js` - 配置检查脚本（关键！）

```javascript
#!/usr/bin/env node

/**
 * S4：配置检查Skill - 实现代码
 *
 * 作用：在CodeReview和发版前自动检查配置文件
 * 防止：PDP2D_SIZE这样的拼写错误导致白屏
 *
 * 使用：
 *   npm run skill:s4:check-config
 *   或
 *   node scripts/skill-s4-check-config.js
 */

const fs = require('fs');
const path = require('path');

class ConfigChecker {
  constructor() {
    this.errors = [];
    this.warnings = [];
    this.checks = {
      spellingErrors: 0,
      typeErrors: 0,
      sensitiveInfoLeaks: 0,
      envVarIssues: 0
    };
  }

  /**
   * 检查1：拼写错误
   * 扫描所有文件中的环境变量引用，检查是否有typo
   */
  checkSpelling() {
    console.log('\n📝 检查1：拼写错误...');

    // 定义预期的环境变量清单
    const expectedEnvVars = [
      'NODE_ENV', 'PORT', 'DB_HOST', 'DB_PORT', 'DB_USER', 'DB_PASSWORD',
      'REDIS_HOST', 'REDIS_PORT', 'REDIS_PASSWORD', 'LOG_LEVEL', 'CACHE_TTL'
    ];

    // 检查 .env 文件
    const envPath = path.join(process.cwd(), '.env');
    if (fs.existsSync(envPath)) {
      const envContent = fs.readFileSync(envPath, 'utf-8');
      const envVars = envContent.split('\n').map(line => line.split('=')[0].trim()).filter(Boolean);

      // 寻找可能的拼写错误（比如REDIS_PRT而不是REDIS_PORT）
      for (const envVar of envVars) {
        if (!expectedEnvVars.includes(envVar)) {
          const similarity = this.calculateSimilarity(envVar, expectedEnvVars);
          if (similarity.score > 0.8) {
            this.errors.push(`
🔴 【拼写错误】.env 文件中发现可能的环境变量名拼写错误
  发现: ${envVar}
  建议: ${similarity.closest}
  影响: 这个环境变量可能无法正确读取，导致功能异常
            `);
            this.checks.spellingErrors++;
          }
        }
      }
    }
  }

  /**
   * 检查2：敏感信息硬编码
   * 扫描配置文件，检查是否硬编码了密码、API Key等
   */
  checkSensitiveInfo() {
    console.log('📝 检查2：敏感信息泄露...');

    const sensitivePatterns = [
      { pattern: /password\s*:\s*['"].*['"]/, key: 'password', severity: '高' },
      { pattern: /api[_-]?key\s*:\s*['"].*['"]/, key: 'API Key', severity: '高' },
      { pattern: /secret\s*:\s*['"].*['"]/, key: 'secret', severity: '高' },
      { pattern: /token\s*:\s*['"].*['"]/, key: 'token', severity: '中' }
    ];

    const configFiles = [
      'src/config/index.ts',
      'src/config/database.ts',
      'src/config/redis.ts',
      '.env'
    ];

    for (const file of configFiles) {
      const filePath = path.join(process.cwd(), file);
      if (!fs.existsSync(filePath)) continue;

      const content = fs.readFileSync(filePath, 'utf-8');
      const lines = content.split('\n');

      for (let i = 0; i < lines.length; i++) {
        const line = lines[i];

        for (const { pattern, key, severity } of sensitivePatterns) {
          if (pattern.test(line) && !line.includes('process.env')) {
            this.errors.push(`
🔴 【安全风险】${file} 第 ${i + 1} 行：${key} 硬编码为明文
  内容: ${line.trim()}
  应改为: ${key.toLowerCase()}: process.env.${key.toUpperCase()}_PASSWORD
  严重程度: ${severity}
            `);
            this.checks.sensitiveInfoLeaks++;
          }
        }
      }
    }
  }

  /**
   * 检查3：环境变量引用
   * 检查所有 process.env.XXX 是否真的在环境中定义
   */
  checkEnvVarReferences() {
    console.log('📝 检查3：环境变量引用检查...');

    // 读取 .env 文件中定义的变量
    const envPath = path.join(process.cwd(), '.env');
    const definedVars = new Set();

    if (fs.existsSync(envPath)) {
      const envContent = fs.readFileSync(envPath, 'utf-8');
      envContent.split('\n').forEach(line => {
        const key = line.split('=')[0].trim();
        if (key) definedVars.add(key);
      });
    }

    // 扫描代码中引用的环境变量
    const srcPath = path.join(process.cwd(), 'src');
    const referencedVars = new Set();
    const envPattern = /process\.env\.([A-Z_0-9]+)/g;

    const scanDir = (dir) => {
      if (!fs.existsSync(dir)) return;
      const files = fs.readdirSync(dir);

      for (const file of files) {
        const filePath = path.join(dir, file);
        const stat = fs.statSync(filePath);

        if (stat.isDirectory()) {
          scanDir(filePath);
        } else if (file.endsWith('.ts') || file.endsWith('.js')) {
          const content = fs.readFileSync(filePath, 'utf-8');
          let match;
          while ((match = envPattern.exec(content)) !== null) {
            referencedVars.add(match[1]);
          }
        }
      }
    };

    scanDir(srcPath);

    // 检查是否有未定义的环境变量
    for (const varName of referencedVars) {
      if (!definedVars.has(varName)) {
        this.warnings.push(`
🟡 【环境变量未定义】代码中引用了 ${varName}，但在 .env 中未定义
  建议: 在 .env 中添加 ${varName}=<value>
        `);
        this.checks.envVarIssues++;
      }
    }
  }

  /**
   * 检查4：配置值范围
   * 检查配置值是否在预期范围内
   */
  checkConfigRanges() {
    console.log('📝 检查4：配置值范围检查...');

    const port = parseInt(process.env.PORT || '3000');
    const redisPort = parseInt(process.env.REDIS_PORT || '6379');
    const cacheTTL = parseInt(process.env.CACHE_TTL || '3600');

    // 检查端口
    if (port < 1 || port > 65535) {
      this.errors.push(`
🔴 【范围错误】PORT=${port} 不在有效范围（1-65535）
  当前值: ${port}
  应改为: 1000-65535 之间的值
      `);
    }

    // 检查Redis端口
    if (redisPort < 1 || redisPort > 65535) {
      this.errors.push(`
🔴 【范围错误】REDIS_PORT=${redisPort} 不在有效范围（1-65535）
  当前值: ${redisPort}
  应改为: 1000-65535 之间的值
      `);
    }

    // 检查缓存TTL
    if (cacheTTL < 60 || cacheTTL > 86400) {
      this.warnings.push(`
🟡 【范围警告】CACHE_TTL=${cacheTTL} 不在建议范围（60-86400）
  当前值: ${cacheTTL}秒
  建议范围: 60-86400秒（1分钟 - 1天）
  说明: 太短可能导致缓存失效，太长可能导致数据不新鲜
      `);
    }
  }

  /**
   * 辅助函数：计算字符串相似度
   */
  calculateSimilarity(str, options) {
    let closest = null;
    let bestScore = 0;

    for (const option of options) {
      const score = this.levenshteinSimilarity(str, option);
      if (score > bestScore) {
        bestScore = score;
        closest = option;
      }
    }

    return { closest, score: bestScore };
  }

  levenshteinSimilarity(s1, s2) {
    const longer = s1.length > s2.length ? s1 : s2;
    const shorter = s1.length > s2.length ? s2 : s1;

    if (longer.length === 0) return 1.0;

    const editDistance = this.levenshteinDistance(longer, shorter);
    return (longer.length - editDistance) / longer.length;
  }

  levenshteinDistance(s1, s2) {
    const costs = [];
    for (let i = 0; i <= s1.length; i++) {
      let lastValue = i;
      for (let j = 0; j <= s2.length; j++) {
        if (i === 0) {
          costs[j] = j;
        } else if (j > 0) {
          let newValue = costs[j - 1];
          if (s1.charAt(i - 1) !== s2.charAt(j - 1)) {
            newValue = Math.min(Math.min(newValue, lastValue), costs[j]) + 1;
          }
          costs[j - 1] = lastValue;
          lastValue = newValue;
        }
      }
      if (i > 0) costs[s2.length] = lastValue;
    }
    return costs[s2.length];
  }

  /**
   * 运行所有检查
   */
  run() {
    console.log('\n═══════════════════════════════════════════');
    console.log('🔍 S4：配置检查 Skill - 开始运行');
    console.log('═══════════════════════════════════════════\n');

    this.checkSpelling();
    this.checkSensitiveInfo();
    this.checkEnvVarReferences();
    this.checkConfigRanges();

    this.report();
  }

  /**
   * 生成报告
   */
  report() {
    console.log('\n═══════════════════════════════════════════');
    console.log('📊 检查结果');
    console.log('═══════════════════════════════════════════\n');

    console.log(`✅ 检查项统计：`);
    console.log(`  拼写错误: ${this.checks.spellingErrors}`);
    console.log(`  敏感信息泄露: ${this.checks.sensitiveInfoLeaks}`);
    console.log(`  环境变量引用问题: ${this.checks.envVarIssues}`);

    if (this.errors.length > 0) {
      console.log('\n🔴 发现严重错误（必须修复）：');
      this.errors.forEach(err => console.log(err));
    }

    if (this.warnings.length > 0) {
      console.log('\n🟡 发现警告（建议修复）：');
      this.warnings.forEach(warn => console.log(warn));
    }

    const canDeploy = this.errors.length === 0;
    console.log('\n═══════════════════════════════════════════');
    if (canDeploy) {
      console.log('✅ 配置检查通过，可以继续部署');
      console.log('═══════════════════════════════════════════\n');
      process.exit(0);
    } else {
      console.log('❌ 配置检查失败，必须修复所有错误后才能部署');
      console.log('═══════════════════════════════════════════\n');
      process.exit(1);
    }
  }
}
```
